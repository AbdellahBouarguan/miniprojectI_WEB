
\chapter*{Liste des abréviations}
\addcontentsline{toc}{chapter}{Liste des abréviations}

\begin{description}[leftmargin=3.2cm,style=nextline]
    \item[API] Interface de programmation applicative.
    \item[CRUD] Create, Read, Update, Delete (créer, lire, mettre à jour et supprimer).
    \item[CSRF] Cross-Site Request Forgery (falsification de requête intersites).
    \item[MVC] Modèle--Vue--Contrôleur.
    \item[PDO] PHP Data Objects.
    \item[SGBD] Système de gestion de base de données.
    \item[SQL] Structured Query Language (langage de requête structuré).
    \item[TVA] Taxe sur la valeur ajoutée.
    \item[UC] Cas d'utilisation.
    \item[XSS] Cross-Site Scripting (injection de scripts intersites).
\end{description}
